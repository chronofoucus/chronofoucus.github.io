/* ================================ ChronoFocus Daily Goals Script ================================ */ /* ===== LOAD GOALS ON PAGE LOAD ===== */ document.addEventListener("DOMContentLoaded", loadGoals); function loadGoals() { let goals = JSON.parse(localStorage.getItem("goals")) || []; goals.forEach(goal => { createGoalElement(goal.text, goal.completed); }); } /* ===== ADD GOAL ===== */ function addGoal() { let input = document.getElementById("goalInput"); let goalText = input.value.trim(); if (goalText === "") { alert("Please enter a goal!"); return; } createGoalElement(goalText, false); saveGoal(goalText, false); input.value = ""; } /* ===== CREATE GOAL ELEMENT ===== */ function createGoalElement(text, completed) { let goalList = document.getElementById("goalList"); let li = document.createElement("li"); li.style.margin = "10px 0"; li.style.display = "flex"; li.style.justifyContent = "space-between"; li.style.alignItems = "center"; let span = document.createElement("span"); span.textContent = text; span.style.cursor = "pointer"; if (completed) { span.style.textDecoration = "line-through"; span.style.opacity = "0.6"; } /* ===== MARK GOAL COMPLETE ===== */ span.onclick = function () { if (!span.classList.contains("completed")) { span.style.textDecoration = "line-through"; span.style.opacity = "0.6"; span.classList.add("completed"); updateGoalStatus(text, true); // Add 15 reward points addRewardPoints(15); alert("Goal completed! +15 Reward Points 🎯💜"); } }; /* ===== DELETE BUTTON ===== */ let deleteBtn = document.createElement("button"); deleteBtn.textContent = "Delete"; deleteBtn.className = "main-btn"; deleteBtn.style.padding = "6px 14px"; deleteBtn.style.fontSize = "12px"; deleteBtn.onclick = function () { li.remove(); deleteGoal(text); }; li.appendChild(span); li.appendChild(deleteBtn); goalList.appendChild(li); } /* ===== SAVE GOAL ===== */ function saveGoal(text, completed) { let goals = JSON.parse(localStorage.getItem("goals")) || []; goals.push({ text: text, completed: completed }); localStorage.setItem("goals", JSON.stringify(goals)); } /* ===== UPDATE GOAL STATUS ===== */ function updateGoalStatus(text, status) { let goals = JSON.parse(localStorage.getItem("goals")) || []; goals = goals.map(goal => { if (goal.text === text) { goal.completed = status; } return goal; }); localStorage.setItem("goals", JSON.stringify(goals)); } /* ===== DELETE GOAL ===== */ function deleteGoal(text) { let goals = JSON.parse(localStorage.getItem("goals")) || []; goals = goals.filter(goal => goal.text !== text); localStorage.setItem("goals", JSON.stringify(goals)); }
